    <script>
      // WebGPU Detection
      (function() {
        const browserInfo = document.getElementById('browser-info');
        const webgpuError = document.getElementById('webgpu-error');

        // Get browser info
        const ua = navigator.userAgent;
        const browserName = (function() {
          if (ua.includes('Chrome')) return 'Chrome';
          if (ua.includes('Firefox')) return 'Firefox';
          if (ua.includes('Safari') && !ua.includes('Chrome')) return 'Safari';
          if (ua.includes('Edge')) return 'Edge';
          if (ua.includes('Opera')) return 'Opera';
          return 'Unknown';
        })();

        browserInfo.textContent = `Browser: ${browserName} | User Agent: ${ua.substring(0, 100)}...`;

        // Check for WebGPU support
        if (!navigator.gpu) {
          webgpuError.classList.add('visible');
          console.error('WebGPU is not supported in this browser');
          return;
        }

        // Additional check: try to request adapter
        navigator.gpu.requestAdapter().then(adapter => {
          if (!adapter) {
            webgpuError.classList.add('visible');
            browserInfo.innerHTML += '<br><br>WebGPU API exists but no adapter found. This may be due to:<br>- Disabled hardware acceleration<br>- Unsupported GPU<br>- WebGPU disabled in flags';
            console.error('WebGPU adapter not available');
          } else {
            console.log('WebGPU is supported:', adapter);
          }
        }).catch(err => {
          webgpuError.classList.add('visible');
          browserInfo.innerHTML += `<br><br>Error requesting WebGPU adapter: ${err.message}`;
          console.error('Error requesting WebGPU adapter:', err);
        });
      })();
    </script>

    <script>
      // Surface runtime errors directly in the page to help diagnose missing UI
      window.addEventListener('error', e => {
        try {
          const container = document.getElementById('app') || document.body;
          const msg = (e && e.message) ? e.message : String(e.error || e);
          const pre = document.createElement('pre');
          pre.style.color = 'white';
          pre.style.background = '#ff6b6b';
          pre.style.padding = '12px';
          pre.style.borderRadius = '6px';
          pre.textContent = `Runtime error: ${msg}`;
          container.prepend(pre);
          console.error('Captured runtime error:', e);
        } catch (err) {
          console.error('Failed to render runtime error to page:', err);
        }
      });
      window.addEventListener('unhandledrejection', e => {
        try {
          const container = document.getElementById('app') || document.body;
          const pre = document.createElement('pre');
          pre.style.color = 'white';
          pre.style.background = '#ff6b6b';
          pre.style.padding = '12px';
          pre.style.borderRadius = '6px';
          pre.textContent = `Unhandled Rejection: ${String(e.reason)}`;
          container.prepend(pre);
          console.error('Unhandled rejection:', e);
        } catch (err) {
          console.error('Failed to render unhandled rejection to page:', err);
        }
      });
    </script>
